// helpers ---------------------------------------------------------------
function slide(title, section) {
  const s = pres.addSlide({ masterName: "CONTENT", sectionTitle: section });
  s.addText(title, { placeholder: "title" });
  return s;
}
function bullets(s, items, x, y, w, h, size = 16) {
  const arr = items.map((it, i) => {
    const sub = typeof it === "object" && it.sub;
    const t = typeof it === "object" ? it.t : it;
    return { text: t, options: { bullet: sub ? { indent: 18 } : true, indentLevel: sub ? 1 : 0, fontSize: sub ? size - 2 : size, color: sub ? "6B5E4A" : "3E3426", breakLine: i < items.length - 1, paraSpaceAfter: 6 } };
  });
  s.addText(arr, { x, y, w, h, valign: "top", isTextBox: true, margin: 4 });
}
function stat(s, x, y, w, value, label, color = NAVY) {
  s.addShape(pres.shapes.RECTANGLE, { x, y, w, h: 1.35, fill: { color: LIGHT }, line: { color: LIGHT } });
  s.addText(value, { x: x + 0.15, y: y + 0.08, w: w - 0.3, h: 0.7, fontSize: 30, bold: true, color, isTextBox: true, margin: 0 });
  s.addText(label, { x: x + 0.15, y: y + 0.78, w: w - 0.3, h: 0.5, fontSize: 12, color: "6B5E4A", isTextBox: true, margin: 0, valign: "top" });
}
function table(s, rows, x, y, w, colW, opts = {}) {
  const fs = opts.fontSize || 13;
  const data = rows.map((r, i) => r.map((c, j) => ({
    text: String(c),
    options: i === 0 ? { bold: true, color: "FFFFFF", fill: { color: NAVY }, fontSize: fs, align: (j === 0 || !opts.numeric) ? "left" : "center" }
      : { color: "3E3426", fill: { color: i % 2 ? ROW : LIGHT }, fontSize: fs, align: (j === 0 || !opts.numeric) ? "left" : "center", bold: opts.boldRow === i },
  })));
  s.addTable(data, { x, y, w, colW, border: { type: "solid", pt: 0.5, color: GRID }, rowH: opts.rowH || 0.38, valign: "middle", margin: 0.05 });
}
function note(s, text, y = 6.45) {
  s.addText(text, { x: 0.6, y, w: 12.1, h: 0.4, fontSize: 11, italic: true, color: GREY, isTextBox: true, margin: 0 });
}
